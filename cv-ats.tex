%--- Fonts and ligatures -------------------------------------------------------
% The "fi"/"ff"/"ffl" ligatures are single glyphs. Some resume parsers extract
% them as U+FB01/U+FB00/U+FB04 rather than as plain letters, which silently
% breaks keyword matching ("AI-first", "effort", "profiling", "offline").
% We therefore switch ligatures OFF and emit a proper Unicode map, on whichever
% engine is used, so the extracted text is always plain ASCII letters.
\ifPDFTeX
  \usepackage[T1]{fontenc}
  \usepackage[utf8]{inputenc}
  \usepackage{cmap}                % searchable / copy-pasteable PDF text
  \usepackage{lmodern}             % Latin Modern: the Computer Modern look of
                                   % the previous resume, but as scalable Type 1
                                   % outlines. Plain [T1]{fontenc} without this
                                   % can fall back to Type 3 BITMAP fonts, from
                                   % which resume parsers extract nothing at all.
  \input{glyphtounicode}           % map every glyph to real Unicode
  \pdfgentounicode=1
  \usepackage{microtype}
  \DisableLigatures[f]{encoding = *, family = *}
\else
  \usepackage{fontspec}
  \setmainfont{lmroman10}[
    Extension     = .otf,
    UprightFont   = *-regular,
    BoldFont      = *-bold,
    ItalicFont    = *-italic,
    BoldItalicFont= *-bolditalic,
    Ligatures     = NoCommon]
\fi
